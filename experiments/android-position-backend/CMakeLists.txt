cmake_minimum_required(VERSION 3.20)
project(android_position_candidate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(position_candidate STATIC PositionSnapshot.cpp)
target_include_directories(position_candidate PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(ANDROID)
    if(NOT ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_PLATFORM_LEVEL LESS 28)
        message(FATAL_ERROR "The pinned game candidate requires ARM64/API28+")
    endif()
    add_library(android_position_candidate STATIC AndroidSelfMemory.cpp)
    target_link_libraries(android_position_candidate PUBLIC position_candidate dl)
    target_include_directories(android_position_candidate PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../Shared/sdk")
    set_target_properties(android_position_candidate position_candidate PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
endif()
include(CTest)
if(BUILD_TESTING AND NOT ANDROID)
    add_executable(position_candidate_tests tests.cpp)
    target_link_libraries(position_candidate_tests PRIVATE position_candidate)
    add_test(NAME pinned_position_candidate_boundaries COMMAND position_candidate_tests)
endif()
