cmake_minimum_required(VERSION 3.24)
project(mta_experimental_net_module LANGUAGES C CXX)
add_executable(bitbuffer_tests bitbuffer_tests.cpp)
target_compile_features(bitbuffer_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(bitbuffer_tests PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(bitbuffer_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME bitbuffer_contract COMMAND bitbuffer_tests)

add_library(mta_bitstream STATIC MtaBitStream.cpp ModuleRuntime.cpp)
target_compile_features(mta_bitstream PUBLIC cxx_std_20)
target_include_directories(mta_bitstream PUBLIC ../../Shared/sdk ../../Server/sdk)
target_compile_definitions(mta_bitstream PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_definitions(mta_bitstream PUBLIC WIN32)
    target_compile_options(mta_bitstream PUBLIC /utf-8 /Zc:__cplusplus)
endif()
add_executable(mta_adapter_tests mta_adapter_tests.cpp)
target_link_libraries(mta_adapter_tests PRIVATE mta_bitstream)
add_test(NAME mta_bitstream_interface COMMAND mta_adapter_tests)

set_target_properties(mta_bitstream PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(bitstream_boundary SHARED BitStreamFactory.cpp)
target_link_libraries(bitstream_boundary PRIVATE mta_bitstream)
add_executable(module_boundary_tests module_boundary_tests.cpp)
target_link_libraries(module_boundary_tests PRIVATE mta_bitstream ${CMAKE_DL_LIBS})
add_test(NAME mta_bitstream_module_boundary COMMAND module_boundary_tests $<TARGET_FILE:bitstream_boundary>)
set_tests_properties(bitbuffer_contract mta_bitstream_interface mta_bitstream_module_boundary PROPERTIES TIMEOUT 15)
add_executable(protocol_tests protocol_tests.cpp)
target_compile_features(protocol_tests PRIVATE cxx_std_20)
add_test(NAME owned_protocol_frame COMMAND protocol_tests)
set_tests_properties(owned_protocol_frame PROPERTIES TIMEOUT 15)

include(FetchContent)
FetchContent_Declare(enet
    URL https://codeload.github.com/lsalzman/enet/tar.gz/2662c0de09e36f2a2030ccc2c528a3e4c9e8138a
    URL_HASH SHA256=37381831c5ccbb41c3c71e867203fee847670f67717f297f930290d18985ad5f
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(enet)
add_library(net_experimental SHARED OwnedNetServer.cpp)
target_include_directories(net_experimental PRIVATE ${enet_SOURCE_DIR}/include ../../Shared)
target_link_libraries(net_experimental PRIVATE mta_bitstream enet)
if(WIN32)
    target_link_libraries(net_experimental PRIVATE ws2_32 winmm bcrypt)
endif()
add_executable(server_module_tests server_module_tests.cpp)
target_include_directories(server_module_tests PRIVATE ${enet_SOURCE_DIR}/include)
target_link_libraries(server_module_tests PRIVATE mta_bitstream enet ${CMAKE_DL_LIBS})
if(WIN32)
    target_link_libraries(server_module_tests PRIVATE ws2_32 winmm)
endif()
add_test(NAME owned_cnetserver_module COMMAND server_module_tests $<TARGET_FILE:net_experimental>)
set_tests_properties(owned_cnetserver_module PROPERTIES TIMEOUT 40)

target_link_libraries(headless_client PRIVATE mta_bitstream enet)
if(WIN32)
