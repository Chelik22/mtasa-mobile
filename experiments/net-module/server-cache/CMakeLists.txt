cmake_minimum_required(VERSION 3.20)
project(mta_server_cache_filename_contract LANGUAGES CXX)
add_executable(cache_filename_tests cache_filename_tests.cpp)
target_compile_features(cache_filename_tests PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(cache_filename_tests PRIVATE /W4 /MP1)
else()
    target_compile_options(cache_filename_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME server_cache_filename_contract COMMAND cache_filename_tests)
add_test(NAME server_cache_symlinks COMMAND cache_filename_tests links)
set_tests_properties(server_cache_filename_contract server_cache_symlinks PROPERTIES TIMEOUT 15)
set_tests_properties(server_cache_symlinks PROPERTIES SKIP_RETURN_CODE 77)
