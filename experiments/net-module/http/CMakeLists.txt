cmake_minimum_required(VERSION 3.24)
project(mta_owned_http LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Standalone proof requires the installed MSVC Win32 toolchain")
endif()
get_filename_component(MTA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(VENDORED_CURL "${MTA_ROOT}/vendor/curl")
set(VENDORED_ZLIB "${MTA_ROOT}/vendor/zlib")
set(MTA_HTTP_VENDOR_ARTIFACT_DIR "" CACHE PATH "Optional locally built HTTP005 Release curl/zlib artifacts")
if(MTA_HTTP_VENDOR_ARTIFACT_DIR)
    if(CMAKE_CONFIGURATION_TYPES)
        if(NOT CMAKE_CONFIGURATION_TYPES STREQUAL "Release")
            message(FATAL_ERROR "HTTP005 artifact reuse requires CMAKE_CONFIGURATION_TYPES=Release")
        endif()
    elseif(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
        message(FATAL_ERROR "HTTP005 artifact reuse requires CMAKE_BUILD_TYPE=Release")
    endif()
    find_package(Git REQUIRED)
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MTA_ROOT}" diff --exit-code e14c1a603db7512c092ffec6b0d4a60df5351b4f -- vendor/curl vendor/zlib
        RESULT_VARIABLE vendor_diff OUTPUT_QUIET ERROR_QUIET)
    if(NOT vendor_diff EQUAL 0)
        message(FATAL_ERROR "Vendored HTTP inputs differ from the recorded artifact build; use a source build")
    endif()
    execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${MTA_ROOT}" ls-tree -r --name-only e14c1a603db7512c092ffec6b0d4a60df5351b4f -- vendor/curl vendor/zlib
        RESULT_VARIABLE vendor_list_result OUTPUT_VARIABLE vendor_expected)
    if(NOT vendor_list_result EQUAL 0)
        message(FATAL_ERROR "Cannot resolve the recorded vendor source inventory")
    endif()
    string(REGEX REPLACE "\n$" "" vendor_expected "${vendor_expected}")
    string(REPLACE "\n" ";" vendor_expected "${vendor_expected}")
    file(GLOB_RECURSE vendor_actual LIST_DIRECTORIES false RELATIVE "${MTA_ROOT}" "${VENDORED_CURL}/*" "${VENDORED_ZLIB}/*")
    list(SORT vendor_expected)
    list(SORT vendor_actual)
    if(NOT "${vendor_expected}" STREQUAL "${vendor_actual}")
        message(FATAL_ERROR "Extra/missing vendor inputs detected; artifact reuse refused")
    endif()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/verification/http-005/receipt.json" vendor_proof)
    foreach(target owned_http_zlib owned_http_curl)
        set(artifact "${MTA_HTTP_VENDOR_ARTIFACT_DIR}/${target}.lib")
        string(JSON expected_hash GET "${vendor_proof}" artifacts "${target}.lib" sha256)
        file(SHA256 "${artifact}" actual_hash)
        if(NOT actual_hash STREQUAL expected_hash)
            message(FATAL_ERROR "Recorded ${target} artifact hash does not match")
        endif()
        add_library(${target} STATIC IMPORTED GLOBAL)
        set_target_properties(${target} PROPERTIES IMPORTED_LOCATION "${artifact}")
    endforeach()
    target_include_directories(owned_http_curl INTERFACE "${VENDORED_CURL}/include")
    target_compile_definitions(owned_http_curl INTERFACE CURL_STATICLIB)
    target_link_libraries(owned_http_curl INTERFACE owned_http_zlib ws2_32 crypt32 Normaliz secur32 bcrypt ncrypt advapi32 wldap32)
else()
# Mirror the pinned upstream premake source/configuration, without downloading
# or changing vendor files. This is a component build, not an MTA engine build.
file(GLOB_RECURSE curl_sources CONFIGURE_DEPENDS "${VENDORED_CURL}/lib/*.c")
list(REMOVE_ITEM curl_sources "${VENDORED_CURL}/lib/amigaos.c")
file(GLOB zlib_sources CONFIGURE_DEPENDS "${VENDORED_ZLIB}/*.c")
list(REMOVE_ITEM zlib_sources "${VENDORED_ZLIB}/example.c")
add_library(owned_http_zlib STATIC ${zlib_sources})
target_include_directories(owned_http_zlib PUBLIC "${VENDORED_ZLIB}")
target_compile_definitions(owned_http_zlib PRIVATE verbose=-1)
add_library(owned_http_curl STATIC ${curl_sources})
target_include_directories(owned_http_curl PUBLIC "${VENDORED_CURL}/include" PRIVATE "${VENDORED_CURL}/lib" "${VENDORED_ZLIB}")
target_compile_definitions(owned_http_curl PRIVATE BUILDING_LIBCURL HTTP_ONLY USE_ZLIB HAVE_LIBZ HAVE_ZLIB_H HAVE_CONFIG_H
    CURL_DISABLE_LIBCURL_OPTION USE_SCHANNEL USE_WINDOWS_SSPI USE_WIN32_IDN WIN32_LEAN_AND_MEAN NOMINMAX WIN32)
target_compile_definitions(owned_http_curl PUBLIC CURL_STATICLIB)
target_link_libraries(owned_http_curl PUBLIC owned_http_zlib ws2_32 crypt32 Normaliz secur32 bcrypt ncrypt advapi32 wldap32)
endif()
add_library(mta_owned_http STATIC HttpManager.cpp FixtureContentPolicy.cpp)
target_compile_features(mta_owned_http PUBLIC cxx_std_20)
target_include_directories(mta_owned_http PUBLIC "${MTA_ROOT}/Shared/sdk")
target_compile_definitions(mta_owned_http PUBLIC WIN32 WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(mta_owned_http PUBLIC owned_http_curl)
add_executable(http_integration_tests http_integration_tests.cpp)
target_link_libraries(http_integration_tests PRIVATE mta_owned_http)
add_executable(fixture_policy_tests fixture_policy_tests.cpp)
target_link_libraries(fixture_policy_tests PRIVATE mta_owned_http)
foreach(target owned_http_zlib owned_http_curl mta_owned_http http_integration_tests fixture_policy_tests)
    get_target_property(is_imported ${target} IMPORTED)
    if(NOT is_imported)
        target_compile_options(${target} PRIVATE /MP1 /utf-8)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    endif()
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME real_localhost_http COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/http_fixture.py" "$<TARGET_FILE:http_integration_tests>")
set_tests_properties(real_localhost_http PROPERTIES TIMEOUT 75)
add_test(NAME fixture_manifest_policy COMMAND fixture_policy_tests)
set_tests_properties(fixture_manifest_policy PROPERTIES TIMEOUT 15)
