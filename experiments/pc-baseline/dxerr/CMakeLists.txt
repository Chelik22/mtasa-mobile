cmake_minimum_required(VERSION 3.21)
project(scoped_dxerr LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use the reviewed MSVC Win32 configuration")
endif()
if(NOT DEFINED DXERR_SOURCE_ROOT OR NOT DEFINED D3DX_INCLUDE_DIR)
    message(FATAL_ERROR "Explicit verified Microsoft source and D3DX includes are required")
endif()
file(READ "${CMAKE_CURRENT_LIST_DIR}/../dxerr-source.lock.json" source_lock)
string(JSON count LENGTH "${source_lock}" files)
math(EXPR last "${count} - 1")
foreach(index RANGE 0 ${last})
    string(JSON path GET "${source_lock}" files ${index} path)
    string(JSON expected GET "${source_lock}" files ${index} sha256)
    get_filename_component(name "${path}" NAME)
    file(SHA256 "${DXERR_SOURCE_ROOT}/${name}" actual)
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Microsoft DxErr source changed: ${name}")
    endif()
endforeach()
add_library(dxerr STATIC "${DXERR_SOURCE_ROOT}/dxerr.cpp")
target_include_directories(dxerr PUBLIC "${DXERR_SOURCE_ROOT}" PRIVATE "${D3DX_INCLUDE_DIR}")
target_compile_definitions(dxerr PRIVATE UNICODE _UNICODE)
target_compile_options(dxerr PRIVATE /MP1)
set_property(TARGET dxerr PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_features(dxerr PRIVATE cxx_std_17)
add_executable(dxerr_link_probe link_probe.cpp)
target_link_libraries(dxerr_link_probe PRIVATE dxerr user32)
set_property(TARGET dxerr_link_probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
