cmake_minimum_required(VERSION 3.24)
project(mta_lab_paths_contract LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This contract exercises real Windows file locks and environment semantics")
endif()
add_executable(lab_paths_probe lab_paths_probe.cpp)
target_compile_features(lab_paths_probe PRIVATE cxx_std_17)
target_compile_definitions(lab_paths_probe PRIVATE MTA_CLIENT NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
target_include_directories(lab_paths_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../Shared/sdk")
add_library(lab_server_routes OBJECT server_routes.cpp)
target_compile_features(lab_server_routes PRIVATE cxx_std_23)
target_compile_definitions(lab_server_routes PRIVATE MTA_SERVER NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
target_include_directories(lab_server_routes PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../Shared/sdk" "${CMAKE_CURRENT_SOURCE_DIR}/../../../vendor")
if(MSVC)
    target_compile_options(lab_paths_probe PRIVATE /W4 /utf-8 /MP1)
    target_compile_options(lab_server_routes PRIVATE /W3 /utf-8 /MP1 /Zc:__cplusplus)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME lab_paths_filesystem_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_lab_paths.py" "$<TARGET_FILE:lab_paths_probe>")
set_tests_properties(lab_paths_filesystem_contract PROPERTIES TIMEOUT 90)
