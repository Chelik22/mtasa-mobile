cmake_minimum_required(VERSION 3.20)
project(mta_android_jni_input LANGUAGES CXX)

if(NOT TARGET android_input_state)
    add_library(android_input_state STATIC ../android-touch-input/AndroidInputState.cpp)
    target_compile_features(android_input_state PUBLIC cxx_std_20)
    target_include_directories(android_input_state PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../android-touch-input")
endif()
set_target_properties(android_input_state PROPERTIES POSITION_INDEPENDENT_CODE ON)

add_library(mta_input_bridge SHARED InputBridge.cpp)
target_link_libraries(mta_input_bridge PRIVATE android_input_state)
if(NOT ANDROID)
    find_package(JNI REQUIRED)
    target_include_directories(mta_input_bridge PRIVATE ${JNI_INCLUDE_DIRS})
endif()
if(MSVC)
    target_compile_options(mta_input_bridge PRIVATE /W4 /MP1)
    target_compile_options(android_input_state PRIVATE /MP1)
else()
    target_compile_options(mta_input_bridge PRIVATE -Wall -Wextra -Wpedantic)
endif()
