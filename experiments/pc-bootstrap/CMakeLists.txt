cmake_minimum_required(VERSION 3.24)
project(mta_owned_client_bootstrap LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Tests require the real Windows environment and DLL ABI")
endif()
set(MTA_OWNED_NET_DLL "" CACHE FILEPATH "Previously verified owned Win32 client DLL")
enable_testing()
foreach(flavor IN ITEMS release debug)
    add_executable(bootstrap_${flavor}_tests selection_tests.cpp)
    target_compile_features(bootstrap_${flavor}_tests PRIVATE cxx_std_17)
    target_compile_definitions(bootstrap_${flavor}_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_include_directories(bootstrap_${flavor}_tests PRIVATE ../../Shared/sdk)
    if(flavor STREQUAL "debug")
        target_compile_definitions(bootstrap_${flavor}_tests PRIVATE MTA_DEBUG)
    endif()
    if(MSVC)
        target_compile_options(bootstrap_${flavor}_tests PRIVATE /W4 /WX /utf-8 /MP1)
    endif()
    add_test(NAME bootstrap_${flavor}_contract COMMAND bootstrap_${flavor}_tests "${MTA_OWNED_NET_DLL}")
    set_tests_properties(bootstrap_${flavor}_contract PROPERTIES TIMEOUT 20)
endforeach()

# Compile the actual pre-Core caller too, without launching the game or linking
# a substitute proxy. This catches SDK context and public ABI declaration errors.
add_library(proxy_source OBJECT ../../Client/loader-proxy/main.cpp)
target_compile_features(proxy_source PRIVATE cxx_std_23)
target_compile_definitions(proxy_source PRIVATE NOMINMAX _WIN32_WINNT=0x0601)
target_include_directories(proxy_source PRIVATE ../../Shared/sdk ../../vendor)
if(MSVC)
    target_compile_options(proxy_source PRIVATE /W3 /utf-8 /MP1 /Zc:__cplusplus)
endif()
