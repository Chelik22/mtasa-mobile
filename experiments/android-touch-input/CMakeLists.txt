cmake_minimum_required(VERSION 3.20)
project(mta_android_touch_input LANGUAGES CXX)

add_library(android_input_state STATIC AndroidInputState.cpp)
target_compile_features(android_input_state PUBLIC cxx_std_20)
target_include_directories(android_input_state PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
    target_compile_options(android_input_state PRIVATE /W4 /MP1)
else()
    target_compile_options(android_input_state PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(ANDROID)
    add_library(android_input_adapter STATIC AndroidInputAdapter.cpp)
    target_link_libraries(android_input_adapter PUBLIC android_input_state android)
endif()

enable_testing()
add_executable(input_state_tests input_state_tests.cpp)
target_link_libraries(input_state_tests PRIVATE android_input_state)
if(ANDROID)
    target_link_libraries(input_state_tests PRIVATE android_input_adapter)
endif()
add_test(NAME portable_multitouch_traces COMMAND input_state_tests)
set_tests_properties(portable_multitouch_traces PROPERTIES TIMEOUT 15)
