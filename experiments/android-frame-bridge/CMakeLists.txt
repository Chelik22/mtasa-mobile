cmake_minimum_required(VERSION 3.20)
project(mta_android_frame_bridge LANGUAGES CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a" OR ANDROID_PLATFORM_LEVEL LESS 28)
    message(FATAL_ERROR "Pinned game frame telemetry requires Android ARM64/API28+")
endif()
add_library(mta_frame_bridge SHARED FrameTelemetry.cpp)
target_compile_features(mta_frame_bridge PRIVATE cxx_std_17)
target_compile_options(mta_frame_bridge PRIVATE -Wall -Wextra -Wpedantic)
target_link_options(mta_frame_bridge PRIVATE "-Wl,--no-undefined" "-Wl,--exclude-libs,ALL")
target_link_libraries(mta_frame_bridge PRIVATE log)
set_target_properties(mta_frame_bridge PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
